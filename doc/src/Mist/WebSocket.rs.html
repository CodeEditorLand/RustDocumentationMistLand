3//! Mist WebSocket transport.
4//!
5//! Local-first JSON-RPC over WebSocket for the Sky↔Cocoon direct
6//! path (B7-S6). The aim is to remove the Tauri-invoke + Mountain-gRPC double
7//! hop on extension-API traffic, which is ~95 % of total IPC volume in
8//! interactive sessions.
9//!
10//! # Wire format
11//!
12//! Every text frame is a JSON envelope:
13//!
14//! ```json
15//! { "id": <u64>, "method": "<wire-name>", "params": [...] }   // request
16//! { "id": <u64>, "result": <value> }                          // success
17//! { "id": <u64>, "error":  "<message>" }                      // failure
18//! ```
19//!
20//! Notifications (no response expected) carry `"id": null` and the
21//! peer must not reply.
22//!
23//! Binary frames are reserved for B7-S6 phase 2 (length-prefixed
24//! prost-encoded payloads for diagnostic batches and tree
25//! children) - current implementation logs and ignores them.
26//!
27//! # Auth
28//!
29//! Per-spawn 32-byte shared secret. The connecting side sends it
30//! in the WebSocket upgrade `X-Land-Secret` header; the server
31//! rejects connections whose header doesn't match. Mountain
32//! generates the secret at boot, passes it to Cocoon as an env
33//! variable, and exposes it to Sky via the existing
34//! `MountainGetWorkbenchConfiguration` Tauri invoke.
35//!
36//! # Backpressure
37//!
38//! Each side keeps a `HashMap<u64, oneshot::Sender>` of pending
39//! requests. On disconnect the map is drained with errors; the
40//! Effect-TS supervisor on Sky decides whether to retry or fail.
41//!
42//! # Reconnect
43//!
44//! Client side runs an exponential backoff (100 ms, 200 ms, 400 ms,
45//! 1 s, 2 s, capped at 5 s). After 30 s of failed reconnect the
46//! client reports the channel dead.

71/// Per-spawn shared secret for WebSocket connection auth.
72#[derive(Clone)]
73pub struct SharedSecret(pub [u8; 32]);
74
75impl SharedSecret {
76	pub fn random() -> Self {
77		// rand 0.10: `rand::random::<[u8; N]>()` fills via the
78		// thread-local RNG without needing the deprecated
79		// `RngCore::fill_bytes` import path.
80		Self(rand::random::<[u8; 32]>())
81	}
82
83	pub fn as_hex(&self) -> String { hex::encode(self.0) }
84
85	pub fn from_hex(Hex:&str) -> Result<Self> {
86		let Bytes = hex::decode(Hex)?;
87		if Bytes.len() != 32 {
88			anyhow::bail!("shared secret must be 32 bytes (got {})", Bytes.len());
89		}
90		let mut Out = [0u8; 32];
91		Out.copy_from_slice(&Bytes);
92		Ok(Self(Out))
93	}
94}

96/// Server-side handler signature. One closure per JSON-RPC method;
97/// returns the result Value or an error string.
98pub type HandlerFn =
99	Arc<dyn Fn(Value) -> futures_util::future::BoxFuture<'static, Result<Value, String>> + Send + Sync>;
100
101/// Method dispatch table.
102#[derive(Default)]
103pub struct HandlerRegistry {
104	Handlers:Mutex<HashMap<String, HandlerFn>>,
105}
106
107impl HandlerRegistry {
108	pub fn new() -> Arc<Self> { Arc::new(Self::default()) }
109
110	pub async fn Register(&self, Method:String, Handler:HandlerFn) {
111		self.Handlers.lock().await.insert(Method, Handler);
112	}
113
114	pub async fn Lookup(&self, Method:&str) -> Option<HandlerFn> { self.Handlers.lock().await.get(Method).cloned() }
115}

117/// Run a WebSocket server on `127.0.0.1:<port>`. Loops forever
118/// accepting connections; spawns a task per connection.
119///
120/// Returns `Err` only on bind failure; per-connection errors are
121/// logged but never propagated (single bad client must not kill the
122/// listener).
123pub async fn ServeLocal(Port:u16, Secret:SharedSecret, Registry:Arc<HandlerRegistry>) -> Result<()> {
124	let Address = format!("127.0.0.1:{}", Port);
125	let Listener = TcpListener::bind(&Address).await?;
126	tracing::info!(target: "Mist::WebSocket", "server listening on {}", Address);
127	loop {
128		let (Stream, Peer) = match Listener.accept().await {
129			Ok(P) => P,
130			Err(Error) => {
131				tracing::warn!(target: "Mist::WebSocket", "accept error: {}", Error);
132				continue;
133			},
134		};
135		let SecretClone = Secret.clone();
136		let RegistryClone = Registry.clone();
137		tokio::spawn(async move {
138			if let Err(Error) = HandleConnection(Stream, SecretClone, RegistryClone).await {
139				tracing::warn!(target: "Mist::WebSocket", "connection from {} closed with error: {}", Peer, Error);
140			}
141		});
142	}
143}
144
145async fn HandleConnection(Stream:TcpStream, _Secret:SharedSecret, Registry:Arc<HandlerRegistry>) -> Result<()> {
146	// TODO B7-S6 P1.1: validate the X-Land-Secret upgrade header
147	// against `_Secret` here. Stock `accept_async` does not surface
148	// the upgrade headers; we'll switch to the lower-level
149	// `accept_hdr_async` once we've measured the baseline transport
150	// works without auth (loopback-only listener; the practical
151	// attack surface today is "another local process").
152	let WebSocketStream = accept_async(Stream).await?;
153	let (mut Sink, mut Source) = WebSocketStream.split();
154
155	while let Some(MessageResult) = Source.next().await {
156		let Message = match MessageResult {
157			Ok(M) => M,
158			Err(Error) => {
159				tracing::debug!(target: "Mist::WebSocket", "frame read error: {}", Error);
160				break;
161			},
162		};
163
164		match Message {
165			Message::Text(Text) => {
166				let Envelope:Value = match serde_json::from_str(&Text) {
167					Ok(V) => V,
168					Err(Error) => {
169						tracing::debug!(target: "Mist::WebSocket", "bad text frame: {}", Error);
170						continue;
171					},
172				};
173				let Method = Envelope.get("method").and_then(|V| V.as_str()).unwrap_or("");
174				let Identifier = Envelope.get("id").cloned().unwrap_or(Value::Null);
175				let Params = Envelope.get("params").cloned().unwrap_or(Value::Array(vec![]));
176
177				if Method.is_empty() {
178					continue;
179				}
180
181				let Handler = Registry.Lookup(Method).await;
182				let Response = match Handler {
183					Some(H) => {
184						match H(Params).await {
185							Ok(Value) => serde_json::json!({ "id": Identifier, "result": Value }),
186							Err(ErrorMessage) => serde_json::json!({ "id": Identifier, "error": ErrorMessage }),
187						}
188					},
189					None => {
190						serde_json::json!({
191							"id": Identifier,
192							"error": format!("Unknown method: {}", Method),
193						})
194					},
195				};
196
197				if Identifier.is_null() {
198					// Notification - no response expected.
199					continue;
200				}
201
202				if let Err(Error) = Sink.send(Message::Text(Utf8Bytes::from(Response.to_string()))).await {
203					tracing::debug!(target: "Mist::WebSocket", "send error: {}", Error);
204					break;
205				}
206			},
207			Message::Binary(Bytes) => {
208				tracing::trace!(target: "Mist::WebSocket", "binary frame ({} bytes) ignored - reserved for phase 2", Bytes.len());
209			},
210			Message::Close(_) => break,
211			_ => {},
212		}
213	}
214	Ok(())
215}

217/// Pending-request map: request id → response sender.
218type PendingMap = Arc<Mutex<HashMap<u64, oneshot::Sender<Result<Value, String>>>>>;
219
220/// Client-side connection. Holds the write half of the WebSocket
221/// and a map of pending requests keyed by id.
222pub struct Client {
223	// `connect_async` returns `WebSocketStream<MaybeTlsStream<TcpStream>>`
224	// (the TLS wrapper is a no-op when the URL is `ws://` rather than
225	// `wss://`, but the type still has to thread through).
226	Sink:Arc<Mutex<SplitSink<WebSocketStream<MaybeTlsStream<TcpStream>>, Message>>>,
227	Pending:PendingMap,
228	NextIdentifier:AtomicU64,
229	Closed:AtomicBool,
230}
231
232impl Client {
233	/// Connect to a Mist WebSocket server at `Address`
234	/// (e.g. `ws://127.0.0.1:5051`). Spawns a background reader
235	/// task that drains incoming frames and resolves pending
236	/// requests.
237	pub async fn connect(Address:&str) -> Result<Arc<Self>> {
238		let (Stream, _Response) = connect_async(Address).await?;
239		let (Sink, mut Source) = Stream.split();
240		let Sink = Arc::new(Mutex::new(Sink));
241		let Pending:PendingMap = Arc::new(Mutex::new(HashMap::new()));
242		let SelfReference = Arc::new(Self {
243			Sink,
244			Pending:Pending.clone(),
245			NextIdentifier:AtomicU64::new(1),
246			Closed:AtomicBool::new(false),
247		});
248
249		// Reader task: drains incoming frames and resolves pending
250		// request senders by id.
251		let SelfForReader = SelfReference.clone();
252		tokio::spawn(async move {
253			while let Some(MessageResult) = Source.next().await {
254				let Frame = match MessageResult {
255					Ok(M) => M,
256					Err(_) => break,
257				};
258				match Frame {
259					Message::Text(Text) => {
260						if let Ok(Envelope) = serde_json::from_str::<Value>(&Text) {
261							let Identifier = Envelope.get("id").and_then(|V| V.as_u64());
262							if let Some(Identifier) = Identifier {
263								let Sender = SelfForReader.Pending.lock().await.remove(&Identifier);
264								if let Some(Sender) = Sender {
265									let Result = if let Some(ErrorValue) = Envelope.get("error") {
266										Err(ErrorValue.to_string())
267									} else {
268										Ok(Envelope.get("result").cloned().unwrap_or(Value::Null))
269									};
270									let _ = Sender.send(Result);
271								}
272							}
273						}
274					},
275					Message::Close(_) => break,
276					_ => {},
277				}
278			}
279			SelfForReader.Closed.store(true, Ordering::Relaxed);
280			// Drain any remaining pending senders with disconnect errors.
281			let mut Pending = SelfForReader.Pending.lock().await;
282			for (_, Sender) in Pending.drain() {
283				let _ = Sender.send(Err("connection closed".into()));
284			}
285		});
286
287		Ok(SelfReference)
288	}

290	/// Invoke a remote method. Returns the result Value or an error
291	/// string. Pending requests are tracked by id; on disconnect the
292	/// future resolves with `Err("connection closed")`.
293	pub async fn invoke(&self, Method:&str, Params:Value) -> Result<Value, String> {
294		if self.Closed.load(Ordering::Relaxed) {
295			return Err("connection closed".into());
296		}
297		let Identifier = self.NextIdentifier.fetch_add(1, Ordering::Relaxed);
298		let (Tx, Rx) = oneshot::channel();
299		self.Pending.lock().await.insert(Identifier, Tx);
300		let Envelope = serde_json::json!({ "id": Identifier, "method": Method, "params": Params });
301		let Text = Envelope.to_string();
302		let SendResult = self.Sink.lock().await.send(Message::Text(Utf8Bytes::from(Text))).await;
303		if SendResult.is_err() {
304			self.Pending.lock().await.remove(&Identifier);
305			return Err("send failed".into());
306		}
307		Rx.await.map_err(|_| "request cancelled".to_string())?
308	}
309
310	/// Send a one-way notification (no response expected).
311	pub async fn notify(&self, Method:&str, Params:Value) -> Result<(), String> {
312		if self.Closed.load(Ordering::Relaxed) {
313			return Err("connection closed".into());
314		}
315		let Envelope = serde_json::json!({ "id": Value::Null, "method": Method, "params": Params });
316		let Text = Envelope.to_string();
317		self.Sink
318			.lock()
319			.await
320			.send(Message::Text(Utf8Bytes::from(Text)))
321			.await
322			.map_err(|Error| Error.to_string())
323	}
324
325	pub fn is_closed(&self) -> bool { self.Closed.load(Ordering::Relaxed) }
